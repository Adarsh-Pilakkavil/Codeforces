t=int(input())
for _ in range(t):
    n=int(input())
    l=list(map(int,input().split()))
    c1=[0]*(len(l)+1)
    c2=[0]*(len(l)+1)
    for i in range(len(l)):
        if l[i]==1:
            c1[i+1]=c1[i]+1
        else:
            c1[i+1]=c1[i]-1
        if l[i]==3:
            c2[i+1]=c2[i]-1
        else:
            c2[i+1]=c2[i]+1
    m=[float("-inf")]*(len(l)+1)
    m[n-1]=c2[n-1]
    for i in range(len(l)-2,0,-1):
        m[i]=max(m[i+1],c2[i])
    a="NO"
    for x in range(1,len(l)-1):
        if c1[x]>=0 and m[x+1]>=c2[x]:
            a="YES"
            break
    print(a)
