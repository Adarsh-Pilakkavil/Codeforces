t=int(input())
for _ in range(t):
    n,k=map(int,input().split())
    r=n-k
    if r==1:
        print("-1")
        continue
    if n%2==0:
        c0=n//2
        c1=n//2
        r0=(r+1)//2
        r1=r//2
    else:
        c0=n//2+1
        c1=n//2
        r0=(r+1)//2
        r1=r//2
    one=c1-r1
    zero=c0-r0
    if one<0 or zero<0:
        print("-1")
        continue
    parts = []
    for i in range(r0 + r1):
        if i % 2 == 0:
            parts.append("0" * (zero + 1))
            zero = 0
        else:
            parts.append("1" * (one + 1))
            one = 0
 
    print("".join(parts))
