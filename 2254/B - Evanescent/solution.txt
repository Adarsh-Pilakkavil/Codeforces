t=int(input())
for i in range(t):
    n=int(input())
    f=input()
    r=1
    for i in range(1,n):
        if f[i]!=f[i-1]:
            r+=1
    ans=r
    for i in range(1,n-1):
        c=0
        if f[i-1]==f[i+1] and f[i]!=f[i-1]:
            c=-2
        elif f[i]!=f[i-1] and f[i]!=f[i+1]:
            c=-1
        ans=min(ans,r+c)
    print(ans)
