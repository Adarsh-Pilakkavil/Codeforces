t=int(input())
for _ in range(t):
    n,k=map(int,input().split())
    l=list(map(int,input().split()))
    ce=0
    f=False
    c3=3
    c5=5
    c4=4
    for i in l:
        if i%k==0:
            f=True
            break
        if i%2==0:
            ce+=1
        c3=min(c3,3-(i%3))
        c4=min(c4,4-(i%4))
        c5=min(c5,5-(i%5))
    if f or (k==4 and ce>=2):
        print(0)
        continue
    if k==2:
        print(1)
        continue
    if k==3:
        print(c3)
        continue
    if k==4:
        if ce==1:
            print(1)
        else:
            print(min(2,c4))
        continue
    else:
        print(c5)
