t=int(input())
for _ in range(t):
    n=int(input())
    l=list(map(int,input().split()))
    d={}
    c4=[4,16,37,58,89,145,42,20,1]
    for i in l:
        s=0
        while i not in c4:
            s+=1
            t=0
            while i>0:
                t+=(i%10)**2
                i=i//10
            i=t
        for m in range(len(c4)):
            if i==c4[m]:
                break
        if i==1:
            d[8]=d.get(8,0)+1
        else:
            d[(m-s)%8]=d.get((m-s)%8,0)+1
    count=0
    for i in d:
        count+=d[i]*(d[i]-1)//2
    print(count)
