t=int(input())
for _ in range(t):
    n=int(input())
    s=input()
    if s[0]=="1":
        c=0
        for i in range(1,n):
            if s[i]=="0":
                c+=1
        print(c)
    else:
        cl=s[1:].count("1")
        ans=float("inf")
        l,r=0,n-1-cl
        for i in range(1,n):
            ans=min(ans,l+r)
            if s[i]=="0":
                r-=1
            else:
                l+=1
        ans=min(ans,l+r)
        print(ans)
