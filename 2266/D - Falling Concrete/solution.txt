t=int(input())
for _ in range(t):
    n=int(input())
    l=list(map(int,input().split()))
    a=[]
    for i in range(len(l)):
        a.append(l[i]-i)
    c=1
    ans=1
    a.sort()
    for i in range(1,len(a)):
        if a[i]==a[i-1]:
            continue
        if a[i]==a[i-1]+1:
            c+=1
        else:
            ans=max(ans,c)
            c=1
    ans=max(ans,c)
    print(ans)
