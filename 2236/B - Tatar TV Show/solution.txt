t=int(input())
for m in range(t):
    n,k=map(int,input().split())
    s=input()
    l=[]
    for i in s:
        l.append(i)
    for i in range(n-k):
        if l[i]=="1":
            l[i]="0"
            l[i+k]="0" if l[i+k]=="1" else "1"
    if l==["0" for i in range(len(s))]:
        print("YES")
    else:
        print("NO")
